cmake_minimum_required(VERSION 3.22)
project(TinyRadial VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(radial_core src/core/activation.cpp src/core/config.cpp src/core/menu.cpp src/core/shortcut.cpp)
target_include_directories(radial_core PUBLIC src PRIVATE third_party)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE radial_core)
target_include_directories(core_tests PRIVATE third_party)
target_compile_definitions(core_tests PRIVATE SAMPLE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME core_tests COMMAND core_tests)
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  add_executable(TinyRadial WIN32 src/win32/main.cpp src/win32/overlay.cpp src/win32/platform.cpp)
  target_link_libraries(TinyRadial PRIVATE radial_core d2d1 dwrite user32 gdi32 shell32 advapi32 ole32)
  add_executable(InputProbe WIN32 tools/input_probe.cpp)
  target_link_libraries(InputProbe PRIVATE user32 gdi32)
  if(MINGW)
    target_link_options(TinyRadial PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(InputProbe PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(core_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(MSVC)
    target_compile_options(TinyRadial PRIVATE $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL>)
    target_link_options(TinyRadial PRIVATE $<$<CONFIG:Release>:/LTCG> /OPT:REF /OPT:ICF)
  endif()
  install(TARGETS TinyRadial RUNTIME DESTINATION .)
  install(TARGETS InputProbe RUNTIME DESTINATION tools)
  install(FILES tools/measure-idle.ps1 DESTINATION tools)
  install(FILES config.json README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
  install(DIRECTORY profiles DESTINATION .)
  install(DIRECTORY docs DESTINATION .)
endif()
